        // 更新行号
        function updateLineNumbers() {
            const textarea = document.getElementById('editor-text');
            const lineNumbers = document.getElementById('line-numbers');
            const lines = textarea.value.split('\n').length;
            
            let numbers = '';
            for (let i = 1; i <= Math.max(lines, 10); i++) {
                numbers += i + '<br>';
            }
            lineNumbers.innerHTML = numbers;
        }
        
        document.getElementById('editor-text').addEventListener('input', updateLineNumbers);
        updateLineNumbers();
        
        // 上传文件功能 - 纯前端实现
        const uploadArea = document.getElementById('uploadArea');
        const fileInput = document.getElementById('fileInput');
        const uploadProgress = document.getElementById('uploadProgress');
        const progressBar = document.getElementById('progressBar');
        const uploadStatus = document.getElementById('uploadStatus');
        const successMessage = document.getElementById('successMessage');
        const errorMessage = document.getElementById('errorMessage');
        
        uploadArea.addEventListener('click', () => {
            fileInput.click();
        });
        
        uploadArea.addEventListener('dragover', (e) => {
            e.preventDefault();
            uploadArea.style.borderColor = '#58a6ff';
            uploadArea.style.backgroundColor = 'rgba(88, 166, 255, 0.1)';
        });
        
        uploadArea.addEventListener('dragleave', () => {
            uploadArea.style.borderColor = '#30363d';
            uploadArea.style.backgroundColor = 'transparent';
        });
        
        uploadArea.addEventListener('drop', (e) => {
            e.preventDefault();
            uploadArea.style.borderColor = '#30363d';
            uploadArea.style.backgroundColor = 'transparent';
            
            const files = e.dataTransfer.files;
            handleFiles(files);
        });
        
        fileInput.addEventListener('change', (e) => {
            const files = e.target.files;
            handleFiles(files);
        });
        
        // 模拟上传进度
        function simulateUpload(file) {
            return new Promise((resolve) => {
                let progress = 0;
                const interval = setInterval(() => {
                    progress += 10 + Math.random() * 15;
                    if (progress > 100) {
                        progress = 100;
                        clearInterval(interval);
                        resolve(file);
                    }
                    
                    progressBar.style.width = progress + '%';
                    uploadStatus.textContent = `正在上传 ${file.name}... ${Math.round(progress)}%`;
                }, 100);
            });
        }
        
        async function handleFiles(files) {
            if (files.length === 0) return;
            
            // 重置状态
            successMessage.style.display = 'none';
            errorMessage.style.display = 'none';
            uploadProgress.style.display = 'block';
            uploadStatus.textContent = '';
            progressBar.style.width = '0%';
            
            // 处理每个文件
            for (let i = 0; i < files.length; i++) {
                const file = files[i];
                
                try {
                    // 显示上传进度
                    uploadStatus.textContent = `准备上传 ${file.name}...`;
                    
                    // 模拟上传过程
                    await simulateUpload(file);
                    
                    // 如果是文本文件，可以显示在编辑器中
                    if (file.type.startsWith('text/') || file.name.endsWith('.md') || file.name.endsWith('.txt') || 
                        file.name.endsWith('.js') || file.name.endsWith('.json') || file.name.endsWith('.html')) {
                        
                        const reader = new FileReader();
                        reader.onload = function(e) {
                            document.getElementById('editor-text').value = e.target.result;
                            document.getElementById('current-file').textContent = file.name;
                            updateLineNumbers();
                        };
                        reader.readAsText(file);
                    }
                    
                    // 添加到文件列表
                    addFileToList(file);
                    
                } catch (error) {
                    console.error('上传错误:', error);
                    errorMessage.style.display = 'block';
                    uploadProgress.style.display = 'none';
                    return;
                }
            }
            
            // 上传完成
            uploadStatus.textContent = `成功上传 ${files.length} 个文件`;
            setTimeout(() => {
                uploadProgress.style.display = 'none';
                successMessage.style.display = 'block';
                
                // 3秒后隐藏成功消息
                setTimeout(() => {
                    successMessage.style.display = 'none';
                }, 3000);
            }, 500);
        }
        
        function addFileToList(file) {
            const fileList = document.querySelector('.file-list');
            const fileItem = document.createElement('div');
            fileItem.className = 'file-item';
            
            // 根据文件类型选择图标
            let iconClass = 'fa-file';
            if (file.type.startsWith('image/')) iconClass = 'fa-file-image';
            else if (file.type.startsWith('text/')) iconClass = 'fa-file-alt';
            else if (file.type.includes('javascript')) iconClass = 'fa-file-code';
            else if (file.type.includes('json')) iconClass = 'fa-file-code';
            
            fileItem.innerHTML = `
                <i class="fas ${iconClass} file-icon"></i>
                <span>${file.name}</span>
            `;
            
            fileList.appendChild(fileItem);
        }
        
        // 下载文件功能 - 纯前端实现
        function downloadFile(filename) {
            // 获取编辑器内容或创建示例内容
            let content;
            
            if (filename === 'README.md') {
                content = document.getElementById('editor-text').value;
            } else if (filename === 'example.m3u8') {
                content = `#EXTM3U
#EXTINF:-1 tvg-id="ExampleTV" tvg-name="Example Channel" tvg-logo="https://example.com/logo.png" group-title="General",Example Channel
https://example.com/stream.m3u8
#EXTINF:-1 tvg-id="NewsTV" tvg-name="News Channel" tvg-logo="https://example.com/news.png" group-title="News",News Channel
https://example.com/news.m3u8`;
            } else if (filename === 'iptv-tools.zip') {
                // 对于ZIP文件，我们创建模拟的二进制数据
                content = "这是一个模拟的ZIP文件。在实际应用中，这里应该是真实的二进制数据。";
                // 创建下载链接
                const blob = new Blob([content], { type: 'application/zip' });
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = filename;
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                URL.revokeObjectURL(url);
                
                alert(`开始下载: ${filename}`);
                return;
            }
            
            // 创建下载链接
            const blob = new Blob([content], { type: 'text/plain' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = filename;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            
            alert(`开始下载: ${filename}`);
        }
